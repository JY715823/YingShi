### 2.3 SystemMediaRepository.kt 变更

**LocalSystemMediaQueryCache**:
- 新增 `private var storedAtMillis: Long = 0L` 字段
- `peek(context, maxAgeMillis: Long = 0L)`:
  - `maxAgeMillis <= 0` → 不检查时效（保持现有行为）
  - `maxAgeMillis > 0` → 若 `System.currentTimeMillis() - storedAtMillis > maxAgeMillis` 返回 null
- `store(context, items)`: 同时更新 `storedAtMillis = System.currentTimeMillis()`
- `invalidate(context, clearDisk: Boolean = false)`:
  - 始终清内存（memoryItems + storedAtMillis = 0）
  - `clearDisk = true` 时清除磁盘 ItemsKey + storedAtMillis 持久化字段
  - 持久化 storedAtMillis 到 SharedPreferences（key: `stored_at_millis`）

**SystemMediaRepository 接口**:
- `peekCachedMedia(maxAgeMillis: Long = 0L): List<SystemMediaItem>?` — 增加可选参数

**LocalSystemMediaRepository**:
- `peekCachedMedia(maxAgeMillis)`: 调用 `LocalSystemMediaQueryCache.peek(appContext, maxAgeMillis)`
- `loadMedia(forceRefresh)`:
  - 新增可选参数 `maxAgeMillis: Long = 0L`
  - `!forceRefresh && maxAgeMillis > 0` → 检查 `peek(maxAgeMillis)`，未过期则返回
  - 其他逻辑不变

**invalidateSystemMediaMetadataCache**:
- 改签名为 `fun invalidateSystemMediaMetadataCache(context: Context? = null, clearDisk: Boolean = false)`
- 调用 `LocalSystemMediaQueryCache.invalidate(context, clearDisk)`

### 2.4 SystemMediaViewModel.kt 变更

**SystemMediaUiState 状态更新**:

**init 重构** (line 35-50):
```kotlin
init {
    val cachedItems = repository.peekCachedMedia(maxAgeMillis = 5 * 60 * 1000L)
    if (cachedItems.isNullOrEmpty()) {
        refresh()  // 首次加载或缓存过期，触发常规加载
    } else {
        queriedItems = cachedItems
        hasLoadedOnce = true
        publishState(rawItems = cachedItems, selectedFilter = initialFilter, isLoading = false, errorMessage = null)
        startBackgroundRefresh()  // 后台静默刷新（不显示 loading）
    }
    registerContentObserver()

private fun registerContentObserver() {
    contentObserver = SystemMediaContentObserver.create(application) {
        startBackgroundRefresh()
    }
}

private fun startBackgroundRefresh() {
    if (backgroundRefreshJob?.isActive == true) return
    backgroundRefreshJob = viewModelScope.launch {
        _uiState.value = _uiState.value.copy(isBackgroundRefreshing = true)
        runCatching {
            withContext(Dispatchers.IO) {
                repository.loadMedia(forceRefresh = true)
            }
        }.onSuccess { items ->
            queriedItems = items
            hasLoadedOnce = true
            publishState(rawItems = queriedItems, selectedFilter = _uiState.value.selectedFilter, isLoading = false, errorMessage = null)
        }
        _uiState.value = _uiState.value.copy(isBackgroundRefreshing = false)
    }
}

override fun onCleared() {
    super.onCleared()
    contentObserver?.unregister()
}

private fun publishState(
    rawItems: List<SystemMediaItem>,
    selectedFilter: SystemMediaFilter,
    isLoading: Boolean,
    errorMessage: String?,
) {
    val visibleItems = LocalSystemMediaBridgeRepository.applyOverlay(rawItems)
    val isBackgroundRefreshing = _uiState.value.isBackgroundRefreshing
    _uiState.value = SystemMediaUiState(
        isLoading = isLoading,
        isRefreshing = false,  // publishState 总是清掉手动 loading
        isBackgroundRefreshing = isBackgroundRefreshing,
        selectedFilter = selectedFilter,
        allItems = visibleItems,
        filteredItems = visibleItems.applyFilter(selectedFilter),
        errorMessage = errorMessage,
    )
}
```

**refresh(forceRefresh)** 修改:
- 设置 `isRefreshing = true`（手动 loading）
- 完成后清 `isRefreshing`
- 不与 `backgroundRefreshJob` 互斥（手动可中断后台）

**handleBridgeMutation(event)**:

### 2.5 SystemMediaContentObserver.kt 新增

```kotlin
internal class SystemMediaContentObserver(
    private val handler: Handler,
    private val onChangeCallback: () -> Unit,
) : ContentObserver(handler) {

    private val debounceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    private var debounceJob: Job? = null

    override fun onChange(selfChange: Boolean) {
        onChange(selfChange, null)
    }

    override fun onChange(selfChange: Boolean, uri: Uri?) {
        debounceJob?.cancel()
        debounceJob = debounceScope.launch {
            delay(500L)  // debounce 500ms
            onChangeCallback()
        }
    }

    fun unregister() {
        debounceJob?.cancel()
        debounceScope.cancel()
    }

    companion object {
        fun create(context: Context, onChange: () -> Unit): SystemMediaContentObserver? {
            val handler = Handler(Looper.getMainLooper())
            val observer = SystemMediaContentObserver(handler, onChange)
            val uri = MediaStore.Files.getContentUri("external")
            return try {
                context.applicationContext.contentResolver.registerContentObserver(uri, true, observer)
                observer
            } catch (e: SecurityException) {
                null
            }
        }
    }
}
```

### 5.4 SystemMediaScrubber.kt 变更

- 复杂度从 O(N) → O(log N)

伪代码:
```kotlin
internal fun calculateSystemMediaScrollProgress(
    gridState: LazyGridState,
    gridBlocks: List<SystemMediaGridBlock>,
): Float {
    val visibleIndex = gridState.firstVisibleItemIndex
    val totalBlocks = gridBlocks.size
    if (totalBlocks <= 1) return 0f
    // 二分查找 gridBlocks 中第一个 block.index > visibleIndex 的位置
    var lo = 0
    var hi = totalBlocks
    while (lo < hi) {
        val mid = (lo + hi) ushr 1
        if (gridBlocks[mid].firstItemIndex <= visibleIndex) {
            lo = mid + 1
        } else {
            hi = mid
        }
    }
    val currentIndex = (lo - 1).coerceAtLeast(0)
    return currentIndex.toFloat() / (totalBlocks - 1).coerceAtLeast(1)
}
```

**注意**: 实施时需检查 `SystemMediaGridBlock.firstItemIndex` 字段是否存在。若不存在，使用 `gridBlocks` 的 index（线性查找项较少时可保留线性，二分查找为优化保留）。
