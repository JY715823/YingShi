```kotlin
@Composable
internal fun RealTrashCountdownBadge(
    remainingMillis: Long,
    modifier: Modifier = Modifier,
) {
    // 计算剩余比例 ratio = remainingMillis / 24h
    // ratio > 0.5: 普通态 (softGreenContainer → glowWash 渐变 + viewerOverlayBorder 边线)
    // 0.2..0.5: 警告态 (goldAccent 边线)
    // < 0.2: 接近过期 (memoryContainer → memoryWash 渐变 + destructive 边线 + destructive 文字)
    // Surface(radius.capsule) + drawBehind linearGradient + shadowElevation=4.dp
    // 内含 Icon(Schedule) + Text(剩余时长)
}
```
